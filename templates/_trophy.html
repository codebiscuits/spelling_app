{#- Trophy artwork, drawn as inline SVG. trophy(label, kind, won, size, title) label: the number for kind "table" (shown on the cup), or the list name for kind "list" (shown on the plaque) kind: "table" = gold cup, "list" = silver cup with a ruby star won: false draws the same shape as a pale grey outline size: width in pixels title: accessible name, also used as the SVG title The metal colours are fixed on purpose. They must not use the palette variables, so a trophy looks the same in all seven weekday palettes. Each instance takes a unique gradient id from trophy_uid(), because many SVGs on one page share a single id namespace. -#} {% macro trophy(label, kind="table", won=true, size=96, title="") -%} {%- set u = "tr" ~ trophy_uid() -%} {%- set label = label | string -%} {%- set is_table = (kind == "table") -%} {%- if won -%} {%- if is_table -%} {%- set m0, m1, m2, m3, mline = "#fff6c2", "#f7cd4a", "#c58f12", "#8a5a05", "#7a4e04" -%} {%- else -%} {%- set m0, m1, m2, m3, mline = "#ffffff", "#dde1e7", "#9ea7b3", "#667080", "#4d5663" -%} {%- endif -%} {%- set body = "url(#" ~ u ~ "b)" -%} {%- set metal = "url(#" ~ u ~ "m)" -%} {%- set stroke = mline -%} {%- set plaque = "#3a2719" -%} {%- set plaque_text = "#ffe9a8" if is_table else "#f2f4f7" -%} {%- set number_fill = "#5c3a02" if is_table else "#3d4552" -%} {%- set number_edge = "#fff3b8" if is_table else "#ffffff" -%} {%- set opacity = 1 -%} {%- else -%} {%- set body = "#eceef0" -%} {%- set metal = "#eceef0" -%} {%- set stroke = "#aeb4ba" -%} {%- set plaque = "#f3f4f5" -%} {%- set plaque_text = "#8b9298" -%} {%- set number_fill = "#8b9298" -%} {%- set number_edge = "none" -%} {%- set opacity = 0.8 -%} {%- endif -%} {%- endmacro %}
